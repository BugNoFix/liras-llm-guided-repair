// used in all the actions
bool same_area(double sx, double sy, double dx, double dy) {
    int i;
    bool x_s_in, y_s_in, x_d_in, y_d_in;
    double x_min, x_max, y_min, y_max;

    for (i = 0; i &lt; N_A; i++)
    {
        x_min = FLOOR[i][0][0] &lt; FLOOR[i][2][0] ? FLOOR[i][0][0] : FLOOR[i][2][0];
        x_max = FLOOR[i][0][0] &gt; FLOOR[i][2][0] ? FLOOR[i][0][0] : FLOOR[i][2][0];
        y_min = FLOOR[i][0][1] &lt; FLOOR[i][2][1] ? FLOOR[i][0][1] : FLOOR[i][2][1];
        y_max = FLOOR[i][0][1] &gt; FLOOR[i][2][1] ? FLOOR[i][0][1] : FLOOR[i][2][1];

        x_s_in = (sx &gt;= x_min - 1.0) &amp;&amp; (sx &lt;= x_max + 1.0);
        y_s_in = (sy &gt;= y_min - 1.0) &amp;&amp; (sy &lt;= y_max + 1.0);
        x_d_in = (dx &gt;= x_min - 1.0) &amp;&amp; (dx &lt;= x_max + 1.0);
        y_d_in = (dy &gt;= y_min - 1.0) &amp;&amp; (dy &lt;= y_max + 1.0);

        if ((x_s_in &amp;&amp; y_s_in) &amp;&amp; (x_d_in &amp;&amp; y_d_in)) // start and dest in same area
            return true;
    }
    return false;
}

double pt_dist(double x1, double x2, double y1, double y2) {
    double xdifference = x2-x1;
    double ydifference = y2-y1;
    return sqrt((xdifference*xdifference)+(ydifference*ydifference));
}

/**
 * Checks if the given coordinates (x, y) are inside ANY of the valid floor areas.
 * Returns true if the position is valid, false if it is in a forbidden area.
 */
const double EPSILON = 1.0;

bool isValidPosition(double x, double y) {
    int i;
    double x_min, x_max, y_min, y_max;
    
    for (i = 0; i &lt; N_A; i++) {
        x_min = FLOOR[i][0][0] &lt; FLOOR[i][2][0] ? FLOOR[i][0][0] : FLOOR[i][2][0];
        x_max = FLOOR[i][0][0] &gt; FLOOR[i][2][0] ? FLOOR[i][0][0] : FLOOR[i][2][0];
        y_min = FLOOR[i][0][1] &lt; FLOOR[i][2][1] ? FLOOR[i][0][1] : FLOOR[i][2][1];
        y_max = FLOOR[i][0][1] &gt; FLOOR[i][2][1] ? FLOOR[i][0][1] : FLOOR[i][2][1];
        
        if (x &gt;= x_min - 1.0 &amp;&amp; x &lt;= x_max + 1.0 &amp;&amp; y &gt;= y_min - 1.0 &amp;&amp; y &lt;= y_max + 1.0) {
            return true;
        }
    }
    return false;
}
